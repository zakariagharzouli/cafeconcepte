   155→export function applyBackgroundPresentation(
   156→  frame: HTMLIFrameElement | null,
   157→  backgroundCanvasSelector?: string,
   158→  backgroundVisualSelector?: string,
   159→) {
   160→  const document = frame?.contentDocument;
   161→  if (!document) return;
   162→
   163→  document.getElementById(BACKGROUND_PRESENTATION_STYLE_ID)?.remove();
   164→  document.querySelectorAll("[data-threeui-background-layer]").forEach((element) => {
   165→    element.removeAttribute("data-threeui-background-layer");
   166→    element.removeAttribute("data-threeui-background-fill");
   167→  });
   168→
   169→  if (!backgroundCanvasSelector) {
   170→    document.documentElement.removeAttribute("data-threeui-presentation");
   171→    return;
   172→  }
   173→
   174→  const canvas = document.querySelector<HTMLElement>(backgroundCanvasSelector);
   175→  if (!canvas) return;
   176→
   177→  canvas.setAttribute("data-threeui-background-layer", "");
   178→  canvas.setAttribute("data-threeui-background-fill", "");
   179→  if (backgroundVisualSelector) {
   180→    document.querySelectorAll<HTMLElement>(backgroundVisualSelector).forEach((element) => {
   181→      element.setAttribute("data-threeui-background-layer", "");
   182→    });
   183→  }
   184→
   185→  document.documentElement.setAttribute("data-threeui-presentation", "background");
   186→  const presentationStyle = document.createElement("style");
   187→  presentationStyle.id = BACKGROUND_PRESENTATION_STYLE_ID;
   188→  presentationStyle.textContent = `
   189→    html[data-threeui-presentation="background"],
   190→    html[data-threeui-presentation="background"] body {
   191→      width: 100% !important;
   192→      height: 100% !important;
   193→      min-height: 100% !important;
   194→      overflow: hidden !important;
   195→    }
   196→    html[data-threeui-presentation="background"] body * {
   197→      visibility: hidden !important;
   198→      pointer-events: none !important;
   199→    }
   200→    html[data-threeui-presentation="background"] [data-threeui-background-layer],
   201→    html[data-threeui-presentation="background"] [data-threeui-background-layer] * {
   202→      visibility: visible !important;
   203→    }
   204→    html[data-threeui-presentation="background"] [data-threeui-background-fill] {
   205→      position: fixed !important;
   206→      inset: 0 !important;
   207→      width: 100vw !important;
   208→      height: 100vh !important;
   209→      max-width: none !important;
   210→      max-height: none !important;
   211→      margin: 0 !important;
   212→      transform: none !important;
   213→    }
   214→  `;
   215→  document.head.appendChild(presentationStyle);
   216→
   217→  // The original renderer may have measured a split hero or section-sized
   218→  // canvas. Re-run its authored resize path after the layer becomes full-frame.
   219→  frame.contentWindow?.requestAnimationFrame(() => {
   220→    frame.contentWindow?.dispatchEvent(new Event("resize"));
   221→  });
   222→}

   224→export function LandingPageFrame({
   225→  applyScene,
   226→  backgroundCanvasSelector,
   227→  backgroundVisualSelector,
   228→  className = "",
   229→  customization,
   230→  sourceUrl,
   231→  srcDoc,
   232→  style,
   233→  title,
   234→}: LandingPageFrameProps) {
   235→  const [ready, setReady] = useState(false);
   236→  const frameRef = useRef<HTMLIFrameElement>(null);
   237→
   238→  // Re-applied on every change; the load handler covers the first paint and
   239→  // any navigation the page does inside its own frame.
   240→  useEffect(() => {
   241→    applyPageCustomization(frameRef.current, customization);
   242→    postPageCustomization(frameRef.current, customization);
   243→    applyBackgroundPresentation(frameRef.current, backgroundCanvasSelector, backgroundVisualSelector);
   244→    if (frameRef.current) applyScene?.(frameRef.current);
   245→  }, [applyScene, backgroundCanvasSelector, backgroundVisualSelector, customization]);
   246→
   247→  return (
   248→    <div
   249→      className={`threeui-background landing-page-frame${className ? ` ${className}` : ""}`}
   250→      data-state={ready ? "ready" : "loading"}
   251→      style={{ position: "relative", overflow: "hidden", background: "#080808", pointerEvents: "auto", ...style }}
   252→    >
   253→      <iframe
   254→        ref={frameRef}
   255→        title={title}
   256→        {...(srcDoc ? { srcDoc } : { src: sourceUrl })}
   257→        sandbox={srcDoc ? SRCDOC_FRAME_SANDBOX : URL_FRAME_SANDBOX}
   258→        loading="eager"
   259→        onLoad={(event) => {
   260→          applyPageCustomization(event.currentTarget, customization);
   261→          postPageCustomization(event.currentTarget, customization);
   262→          applyBackgroundPresentation(event.currentTarget, backgroundCanvasSelector, backgroundVisualSelector);
   263→          applyScene?.(event.currentTarget);
   264→          setReady(true);
   265→        }}
   266→        style={{
   267→          position: "absolute",
   268→          inset: 0,
   269→          display: "block",
   270→          width: "100%",
   271→          height: "100%",
   272→          border: 0,
   273→          background: "#080808",
   274→          // A background presentation begins life as the complete source page.
   275→          // Keep that page visually sealed until onLoad has installed the
   276→          // scene-only CSS, otherwise its authored navigation/copy can flash for
   277→          // one paint before the canvas is isolated.
   278→          opacity: backgroundCanvasSelector && !ready ? 0 : 1,
   279→          pointerEvents: backgroundCanvasSelector && !ready ? "none" : "auto",
   280→        }}
   281→      />
   282→    </div>
   283→  );
   284→}

   411→): [BetawiseGlobeSceneProps, Omit<T, keyof BetawiseGlobeSceneProps>] {

   415→/** `undefined` means "leave the document's own value alone". */
   416→function betawiseSceneSettings(scene: BetawiseGlobeSceneProps) {
   417→  const settings: Record<string, number> = {};
   418→  if (scene.tilt !== undefined) settings.TILT_X = scene.tilt;
   419→  if (scene.surfaceContrast !== undefined) settings.SEA_DIM = 1 - scene.surfaceContrast;
   420→  if (scene.limbGlow !== undefined) settings.RIM_LIGHT = scene.limbGlow;
   421→  if (scene.exposure !== undefined) settings.EXPOSURE = scene.exposure;
   422→  if (scene.bloom !== undefined) settings.BLOOM = scene.bloom;
   423→  if (scene.starGlow !== undefined) settings.DUST_GAIN = scene.starGlow;
   424→  if (scene.arcGlow !== undefined) settings.ARC_GAIN = scene.arcGlow;
   425→  return settings;
   426→}

   428→/**
   429→ * Re-identified only when a value actually moves, so the frame effect re-runs on
   430→ * a slider drag and on nothing else. `set()` writes CFG and calls the document's
   431→ * own apply(); TILT_X needs neither, because the frame loop reads it live.
   432→ */
   433→function useSceneSettings(settings: Record<string, number>) {
   434→  const signature = JSON.stringify(settings);
   435→  return useCallback((element: HTMLIFrameElement) => {
   436→    const scene = betawiseScene(element);
   437→    const values = JSON.parse(signature) as Record<string, number>;
   438→    if (scene && Object.keys(values).length > 0) scene.set(values);
   439→  }, [signature]);
   440→}

   442→/** Reads the scene's published hook, which is absent until the frame has run. */
   443→function betawiseScene(element: HTMLIFrameElement) {
   444→  const view = element.contentWindow as { __betawise?: { set: (values: Record<string, number>) => void } } | null;
   445→  return view?.__betawise;
   446→}

   449→ * The authored page is served byte-for-byte for the base entry — the frame loads it
   450→ * from its packaged URL, exactly as it did before the variants existed. The four
   451→ * other worlds are derived: the same authored document with anchored rewrites applied
   452→ * to its tuning block, the surface the stroke field samples, the star volume, and
   453→ * what the arc and marker layers carry. The rewrites live in betawiseGlobeVariants.js
   454→ * so the node test can assert every anchor against the packaged file rather than
   455→ * against a copy of it.

   768→ * The authored file is named ascend.html; the page it holds is Meridian.
   769→ *
   770→ * Earth is served from that URL, exactly as it was before the other bodies
   771→ * existed. The three siblings are derived: the same document with anchored
   772→ * rewrites applied to what the surface shader builds its albedo out of, how
   773→ * much air stands over it, and the colours the frame is graded through —
   774→ * plus, for Saturn, a ring system and the shadow it lays across the disc. The
   775→ * rewrites live in meridianVariants.js so the node test can assert every
   776→ * anchor against the packaged file rather than against a copy of it.
   777→ */
   778→export function MeridianLandingPage({
   779→  variant = "earth",
   780→  presentation = "background",
   781→  orbitSpeed,
   782→  pointerSway,
   783→  sunAzimuth,
   784→  haze,
   785→  halo,
   786→  exposure,
   787→  relief,
   788→  clouds,
   789→  stars,
   790→  rings,
   791→  ...props
   792→}: MeridianLandingPageProps) {
   793→  const safeVariant = MERIDIAN_VARIANTS.includes(variant) ? variant : "earth";
   794→  /* the optional call is not defensiveness: on the public site the builder is a
   795→     stub, and the Pro entry it belongs to is never mounted there */
   796→  const srcDoc = useMemo(() => buildMeridianDocument?.(safeVariant, presentation), [presentation, safeVariant]);
   797→
   798→  /* Rebuilt whenever a slider moves, which is what re-runs the frame effect.
   799→     A knob left undefined never reaches the document, so the authored value
   800→     stands rather than being overwritten by a slider's own default. The three
   801→     that are not uniforms — orbit speed, pointer sway, sun azimuth — need no
   802→     apply at all: the frame loop reads them every frame. */
   803→  const applyScene = useCallback((element: HTMLIFrameElement) => {
   804→    const given = { orbitSpeed, pointerSway, sunAzimuth, haze, halo, exposure, relief, clouds, stars, rings };
   805→    const values: Record<string, number> = {};
   806→    for (const [key, value] of Object.entries(given)) {
   807→      if (typeof value === "number" && Number.isFinite(value)) values[key] = value;
   808→    }
   809→    if (Object.keys(values).length === 0) return;
   810→
   811→    if (safeVariant === "earth") {
   812→      const scene = (element.contentWindow as { __meridian?: { set: (values: Record<string, number>) => void } } | null)?.__meridian;
   813→      scene?.set(values);
   814→      return;
   815→    }
   816→
   817→    /* Derived documents use srcDoc and intentionally keep an opaque sandbox
   818→       origin. Reaching into contentWindow throws there, so their control seam
   819→       travels through the same isolated postMessage bridge as Nocturne. */
   820→    element.contentWindow?.postMessage({ type: "threeui-meridian-controls", values }, "*");
   821→  }, [clouds, exposure, halo, haze, orbitSpeed, pointerSway, relief, rings, safeVariant, stars, sunAzimuth]);
   822→
   823→  return (
   824→    <LandingPageFrame
   825→      {...props}
   826→      key={safeVariant}
   827→      applyScene={applyScene}
   828→      backgroundCanvasSelector={presentation === "background" ? "#scene" : undefined}
   829→      backgroundVisualSelector={presentation === "background" ? "#veil" : undefined}
   830→      title={MERIDIAN_TITLES[safeVariant]}
   831→      sourceUrl="/landing-pages/ascend.html"
   832→      srcDoc={srcDoc}
   833→    />
   834→  );
   835→}

  1271→function normalizeHex(value: string | undefined, fallback: string) {
  1272→  if (typeof value !== "string") return fallback;
  1273→  const match = value.trim().match(/^#([\da-f]{3}|[\da-f]{6})$/i);
  1274→  if (!match) return fallback;
  1275→  const digits = match[1].toLowerCase();
  1276→  return `#${digits.length === 3 ? digits.replace(/./g, (d) => d + d) : digits}`;
  1277→}

  1279→function hexToHsl(hex: string): Hsl {
  1280→  const [red, green, blue] = [1, 3, 5].map((index) => Number.parseInt(hex.slice(index, index + 2), 16) / 255);
  1281→  const max = Math.max(red, green, blue);
  1282→  const min = Math.min(red, green, blue);
  1283→  const l = (max + min) / 2;
  1284→  const delta = max - min;
  1285→  if (delta === 0) return { h: 0, s: 0, l };
  1286→  const s = delta / (1 - Math.abs(2 * l - 1));
  1287→  const base = max === red
  1288→    ? ((green - blue) / delta) % 6
  1289→    : max === green ? (blue - red) / delta + 2 : (red - green) / delta + 4;
  1290→  return { h: (base * 60 + 360) % 360, s, l };
  1291→}

  1293→function hslToRgb({ h, s, l }: Hsl) {
  1294→  const c = (1 - Math.abs(2 * l - 1)) * s;
  1295→  const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
  1296→  const m = l - c / 2;
  1297→  const [r, g, b] = h < 60 ? [c, x, 0]
  1298→    : h < 120 ? [x, c, 0]
  1299→    : h < 180 ? [0, c, x]
  1300→    : h < 240 ? [0, x, c]
  1301→    : h < 300 ? [x, 0, c]
  1302→    : [c, 0, x];
  1303→  return [r + m, g + m, b + m].map((channel) => Math.round(clamp01(channel) * 255));
  1304→}
